"""Build finished cover art and reel backgrounds with title + logo baked on.

AI image generators mangle text, so we let a photo (or AI art) be the
*background* and stamp the title and logo on top as crisp graphics with Pillow.

Three outputs per episode:
  finished_cover(...)   -> 1920x1080 YouTube thumbnail: title + eyebrow + logo
  video_background(...)  -> 1920x1080 video frame: logo only (captions go bottom)
  reel_background(...)   -> 1080x1920 vertical, darkened, small logo (quote added later)
"""

from __future__ import annotations

from pathlib import Path

from PIL import Image, ImageDraw, ImageFilter, ImageFont

ROOT = Path(__file__).resolve().parent.parent
BURGUNDY = (20, 35, 58, 255)
ROSE = (230, 196, 122, 255)

_FONTS = [
    "/System/Library/Fonts/Supplemental/Georgia Bold.ttf",   # macOS
    "/Library/Fonts/Georgia Bold.ttf",
    "/usr/share/fonts/truetype/dejavu/DejaVuSerif-Bold.ttf",  # Linux
]


def _font(size: int) -> ImageFont.FreeTypeFont:
    for f in _FONTS:
        if Path(f).exists():
            return ImageFont.truetype(f, size)
    return ImageFont.load_default()


def _fit(draw, text, max_w, start, min_size=48):
    """Largest font size (from `start`) whose text fits in max_w."""
    size = start
    while size > min_size:
        f = _font(size)
        if draw.textlength(text, font=f) <= max_w:
            return f
        size -= 4
    return _font(min_size)


def _cover(img: Image.Image, w, h) -> Image.Image:
    """Scale-and-crop an image to exactly w x h (fills the frame, may crop)."""
    scale = max(w / img.width, h / img.height)
    r = img.resize((max(1, int(img.width * scale)), max(1, int(img.height * scale))))
    x, y = (r.width - w) // 2, (r.height - h) // 2
    return r.crop((x, y, x + w, y + h))


def _contain(img: Image.Image, w, h) -> Image.Image:
    """Scale an image to fit WHOLLY inside w x h (nothing cropped)."""
    scale = min(w / img.width, h / img.height)
    return img.resize((max(1, int(img.width * scale)), max(1, int(img.height * scale))))


def _framed(bg_path, w, h, blur=45, backdrop_dark=120) -> Image.Image:
    """Show the WHOLE image, fit inside w x h, over a soft blurred fill of itself.

    This never zoom-crops the subject — a portrait photo in a widescreen frame
    (or vice-versa) keeps its full composition, with the empty sides filled by a
    blurred, darkened copy so there are no hard black bars.
    """
    if not (bg_path and Path(bg_path).exists()):
        return Image.new("RGBA", (w, h), BURGUNDY)
    img = Image.open(bg_path).convert("RGBA")
    # If it already matches the frame's aspect (within a hair), just cover it.
    if abs((img.width / img.height) - (w / h)) < 0.02:
        return _cover(img, w, h)
    back = _cover(img, w, h).filter(ImageFilter.GaussianBlur(blur))
    back = Image.alpha_composite(back, Image.new("RGBA", (w, h), (10, 4, 8, backdrop_dark)))
    fg = _contain(img, w, h)
    back.alpha_composite(fg, ((w - fg.width) // 2, (h - fg.height) // 2))
    return back


def _bottom_shade(w, h, strength=210, start=0.45) -> Image.Image:
    grad = Image.new("L", (1, h), 0)
    for i in range(h):
        grad.putpixel((0, i), int(strength * max(0, (i - h * start) / (h * (1 - start)))))
    shade = Image.new("RGBA", (w, h), (10, 4, 8, 255))
    shade.putalpha(grad.resize((w, h)))
    return shade


def _trim(lg: Image.Image) -> Image.Image:
    """Crop to the visible (non-transparent) artwork."""
    bbox = lg.getchannel("A").getbbox()
    return lg.crop(bbox) if bbox else lg


def _key_white_bg(lg: Image.Image) -> Image.Image:
    """If the logo arrives on a solid white background (no real transparency),
    knock that white out so it drops cleanly onto the dark cover. Flood-fills only
    from the four corners, so light areas INSIDE the art (the gold lettering) are
    never touched — and the weathered-stone plate the logo sits on is kept, which
    reads as intentional on the dark artwork. Then trims to the artwork."""
    lg = lg.convert("RGBA")
    if lg.getchannel("A").getextrema()[0] < 245:
        return _trim(lg)  # already has transparency — respect it
    rgb = lg.convert("RGB")
    SENT = (255, 0, 254)
    for seed in ((0, 0), (lg.width - 1, 0), (0, lg.height - 1), (lg.width - 1, lg.height - 1)):
        r, g, b = rgb.getpixel(seed)
        if r > 236 and g > 236 and b > 236:          # only key a white-ish corner
            ImageDraw.floodfill(rgb, seed, SENT, thresh=45)
    src, keyed = lg.getdata(), rgb.getdata()
    lg.putdata([(0, 0, 0, 0) if keyed[i] == SENT else src[i] for i in range(len(src))])
    return _trim(lg)


def _logo(max_w) -> Image.Image | None:
    p = ROOT / "branding" / "logo.png"
    if not p.exists():
        return None
    lg = _key_white_bg(Image.open(p))
    return lg.resize((max_w, max(1, int(max_w * lg.height / lg.width))))


def finished_cover(bg_path, title, show_name, out: Path):
    w, h = 1920, 1080
    cover = _framed(bg_path, w, h)
    cover = Image.alpha_composite(cover, _bottom_shade(w, h))
    d = ImageDraw.Draw(cover)

    tf = _fit(d, title, w - 620, 150)
    tx, ty = 70, h - 90 - (tf.size + 24)
    ef = _font(40)
    d.text((tx + 3, ty - 58), show_name.upper(), font=ef, fill=ROSE)
    d.text((tx + 4, ty + 4), title, font=tf, fill=(0, 0, 0, 190))
    d.text((tx, ty), title, font=tf, fill=(255, 255, 255, 255))

    # Logo top-right, out of the way — the title and eyebrow live at the bottom.
    lg = _logo(340)
    if lg:
        cover.alpha_composite(lg, (w - lg.width - 45, 40))
    out.parent.mkdir(parents=True, exist_ok=True)
    cover.convert("RGB").save(out, quality=92)


def video_background(bg_path, out: Path, title: str = ""):
    """Frame for the video itself — logo top-right, episode title top-left; the
    bottom stays clear for the burned-in captions."""
    w, h = 1920, 1080
    frame = _framed(bg_path, w, h)
    lg = _logo(360)
    if lg:
        frame.alpha_composite(lg, (w - lg.width - 45, 40))

    if title:
        d = ImageDraw.Draw(frame)
        maxw = w - (lg.width if lg else 0) - 220   # leave room for the logo
        ty = 52
        for tf, text in _wrap_title(d, title, maxw, start=76, min_size=40):
            for dx, dy in ((-2, -2), (2, -2), (-2, 2), (2, 2)):
                d.text((60 + dx, ty + dy), text, font=tf, fill=(0, 0, 0, 210))
            d.text((60, ty), text, font=tf, fill=(255, 255, 255, 255))
            ty += tf.size + 6

    out.parent.mkdir(parents=True, exist_ok=True)
    frame.convert("RGB").save(out, quality=92)


def reel_background(bg_path, out: Path, title: str = ""):
    """Vertical background for a reel: darkened, large logo top, episode title
    under it; synced captions are burned on later in the lower third."""
    w, h = 1080, 1920
    frame = _framed(bg_path, w, h)
    dark = Image.new("RGBA", (w, h), (8, 3, 6, 110))
    frame = Image.alpha_composite(frame, dark)
    lg = _logo(420)   # top logo, sized to leave the title and lower-third captions clear
    logo_bottom = 60
    if lg:
        frame.alpha_composite(lg, ((w - lg.width) // 2, 60))
        logo_bottom = 60 + lg.height

    if title:
        d = ImageDraw.Draw(frame)
        ty = logo_bottom + 26
        for line in _wrap_title(d, title, w - 140, start=84):
            tf, text = line
            tx = (w - d.textlength(text, font=tf)) // 2
            for dx, dy in ((-2, -2), (2, -2), (-2, 2), (2, 2)):  # outline for legibility
                d.text((tx + dx, ty + dy), text, font=tf, fill=(0, 0, 0, 210))
            d.text((tx, ty), text, font=tf, fill=(255, 255, 255, 255))
            ty += tf.size + 8

    out.parent.mkdir(parents=True, exist_ok=True)
    frame.convert("RGB").save(out)


def _wrap_title(draw, title, max_w, start=84, min_size=48):
    """Fit the title on one line if it can; otherwise break onto two balanced
    lines. Returns [(font, text), ...]."""
    f = _fit(draw, title, max_w, start, min_size)
    if draw.textlength(title, font=f) <= max_w:
        return [(f, title)]
    words = title.split()
    best = min(range(1, len(words)),
               key=lambda k: abs(len(" ".join(words[:k])) - len(" ".join(words[k:]))))
    l1, l2 = " ".join(words[:best]), " ".join(words[best:])
    f2 = _fit(draw, l1 if len(l1) >= len(l2) else l2, max_w, start, min_size)
    return [(f2, l1), (f2, l2)]
